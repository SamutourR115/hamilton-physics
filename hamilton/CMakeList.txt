add_library(hamilton STATIC
    src/math/vec3.cpp
    src/math/mat3.cpp
    src/math/quaternion.cpp
)

add_library(hamilton::hamilton ALIAS hamilton)

target_include_directories(hamilton
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_compile_features(hamilton PUBLIC cxx_std_17)

if(MSVC)
    target_compile_options(hamilton PRIVATE /W4)
else()
    target_compile_options(hamilton PRIVATE -Wall -Wextra -Wpedantic)
endif()
